IF (@FlagSP = 'FetchPayDataSP')
BEGIN
    ;WITH OfferTotal AS
    (
        SELECT bo.BookingCode,
               po.Discount_rs AS OfferAmount
        FROM GSTtblBooking bo
        LEFT JOIN GSTtblProvideOffer po ON bo.OfferID = po.OfferID
    ),
    ServiceTotal AS
    (
        SELECT X.BookingCode,
               SUM(X.Amount) AS GrandTotal
        FROM
        (
            SELECT b1.BookingCode,
                   (rr.BasePrice * DATEDIFF(DAY, b1.Checkin, b1.Checkout)) AS Amount
            FROM GSTtblBooking b1
            INNER JOIN GSTtblRoom r ON b1.RoomID = r.RoomID
            INNER JOIN GSTtblRoomType rr ON r.RoomTypeID = rr.RoomTypeID
            
            UNION ALL
            SELECT pi.BookingCode,
                   (i.UnitPrice * pi.Quantity)
            FROM GSTtblProvideInventory pi
            INNER JOIN GSTtblInventory i ON pi.InventoryID = i.InventoryID

            UNION ALL
            SELECT ofd.BookingCode,
                   (fm.FoodPrice * ofd.Quantity)
            FROM GSTtblOrderFood ofd
            INNER JOIN GSTtblFoodMenu fm ON ofd.FoodMenuID = fm.FoodMenuID

            UNION ALL
            SELECT bp.BookingCode,
                   bp.Fine
            FROM GSTtblBookingPayment bp
            WHERE bp.Fine > 0

            UNION ALL
             SELECT pe.BookingCode,
                   (e.Charges * pe.NoOfGuest)
            FROM GSTtblProvideEvent pe
            INNER JOIN GSTtblEvent e ON pe.EventID = e.EventID
        ) AS X
        GROUP BY X.BookingCode
    ),
    TotalPaid AS
    (
        SELECT BookingCode, SUM(PaidAmt) AS PaidAmount
        FROM GSTtblBookingPayment
        GROUP BY BookingCode
    )

    SELECT  
        ROW_NUMBER() OVER (ORDER BY ISNULL(bp.InvoiceCode,'') DESC) AS SrNo,
        ISNULL(bp.InvoiceCode,'-') AS InvoiceCode,
        b.BookingCode,
        g.GuestName,
        ISNULL(CONVERT(VARCHAR(10), bp.PaymentDate,103), '-') AS PaymentDate,
        ISNULL(pm.PaymentMethodType,'-') AS PaymentMethodType,
        ISNULL(N'₹' + FORMAT(bp.PaidAmt, 'N2'), '-') AS PaidAmount,
        N'₹' + FORMAT((st.GrandTotal - ISNULL(ot.OfferAmount,0)), 'N2') AS TotalAmount,
        ISNULL(
            N'₹' + FORMAT(
                (st.GrandTotal - ISNULL(ot.OfferAmount,0))  
                - ISNULL((SELECT SUM(p2.PaidAmt)   
                          FROM GSTtblBookingPayment p2   
                          WHERE p2.BookingCode = b.BookingCode    
                            AND p2.InvoiceCode <= ISNULL(bp.InvoiceCode,'')),0), 'N2'
            ), '-') AS RemainingAmount,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0)) = ISNULL(tp.PaidAmount,0)
                THEN 'Paid'
            ELSE 'Pending'
        END AS Status,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0)) = ISNULL(tp.PaidAmount,0)
                THEN 22   -- Paid
            ELSE 4    -- Pending
        END AS StatusID
    FROM GSTtblBooking b
    INNER JOIN GSTtblGuest g ON b.GuestCode = g.GuestCode
    LEFT JOIN GSTtblBookingPayment bp ON b.BookingCode = bp.BookingCode  
    LEFT JOIN GSTtblPaymentMethod pm ON bp.PaymentMethodID = pm.PaymentMethodID
    LEFT JOIN ServiceTotal st ON b.BookingCode = st.BookingCode
    LEFT JOIN OfferTotal ot ON b.BookingCode = ot.BookingCode
    LEFT JOIN TotalPaid tp ON b.BookingCode = tp.BookingCode
    WHERE ISNULL(tp.PaidAmount,0) > 0  
    ORDER BY ISNULL(bp.InvoiceCode,'') DESC;
END

    ;WITH OfferTotal AS
    (
        SELECT bo.BookingCode,
               po.Discount_rs AS OfferAmount
        FROM GSTtblBooking bo
        LEFT JOIN GSTtblProvideOffer po ON bo.OfferID = po.OfferID
    ),
    ServiceTotal AS
    (
        SELECT X.BookingCode,
               SUM(X.Amount) AS GrandTotal
        FROM
        (
            SELECT b1.BookingCode,
                   (rr.BasePrice * DATEDIFF(DAY, b1.Checkin, b1.Checkout)) AS Amount
            FROM GSTtblBooking b1
            INNER JOIN GSTtblRoom r ON b1.RoomID = r.RoomID
            INNER JOIN GSTtblRoomType rr ON r.RoomTypeID = rr.RoomTypeID
            
            UNION ALL
            SELECT pi.BookingCode,
                   (i.UnitPrice * pi.Quantity)
            FROM GSTtblProvideInventory pi
            INNER JOIN GSTtblInventory i ON pi.InventoryID = i.InventoryID

            UNION ALL
            SELECT ofd.BookingCode,
                   (fm.FoodPrice * ofd.Quantity)
            FROM GSTtblOrderFood ofd
            INNER JOIN GSTtblFoodMenu fm ON ofd.FoodMenuID = fm.FoodMenuID

            UNION ALL
            SELECT bp.BookingCode,
                   bp.Fine
            FROM GSTtblBookingPayment bp
            WHERE bp.Fine > 0

            UNION ALL
        SELECT pe.BookingCode,
       (e.Charges * pe.NoOfGuest) AS Charges
FROM GSTtblProvideEvent pe
INNER JOIN GSTtblEvent e ON pe.EventID = e.EventID
        ) AS X
        GROUP BY X.BookingCode
    ),
    TotalPaid AS
    (
        SELECT BookingCode, SUM(PaidAmt) AS PaidAmount
        FROM GSTtblBookingPayment
        GROUP BY BookingCode
    )

    SELECT  
        ROW_NUMBER() OVER (ORDER BY b.BookingCode DESC) AS SrNo,
        '-' AS InvoiceCode,
        b.BookingCode,
        g.GuestName,
        '-' AS PaymentDate,
        '-' AS PaymentMethodType,
        '-' AS PaidAmount,
        N'₹' + FORMAT((st.GrandTotal - ISNULL(ot.OfferAmount,0)), 'N2') AS TotalAmount,
        N'₹' + FORMAT((st.GrandTotal - ISNULL(ot.OfferAmount,0)), 'N2') AS RemainingAmount,
        'Unpaid' AS Status,
        0 AS StatusID
    FROM GSTtblBooking b
    INNER JOIN GSTtblGuest g ON b.GuestCode = g.GuestCode
    LEFT JOIN ServiceTotal st ON b.BookingCode = st.BookingCode
    LEFT JOIN OfferTotal ot ON b.BookingCode = ot.BookingCode
    LEFT JOIN TotalPaid tp ON b.BookingCode = tp.BookingCode
    WHERE ISNULL(tp.PaidAmount,0) = 0  
    ORDER BY b.BookingCode DESC;
END

    (
        SELECT X.BookingCode,
               SUM(X.Amount) AS GrandTotal
        FROM
        (
            SELECT b1.BookingCode,
                   (rr.BasePrice * DATEDIFF(DAY, b1.Checkin, b1.Checkout)) AS Amount
            FROM GSTtblBooking b1
            INNER JOIN GSTtblRoom r ON b1.RoomID = r.RoomID
            INNER JOIN GSTtblRoomType rr ON r.RoomTypeID = rr.RoomTypeID
            
            UNION ALL
            SELECT pi.BookingCode,
                   (i.UnitPrice * pi.Quantity)
            FROM GSTtblProvideInventory pi
            INNER JOIN GSTtblInventory i ON pi.InventoryID = i.InventoryID

            UNION ALL
            SELECT ofd.BookingCode,
                   (fm.FoodPrice * ofd.Quantity)
            FROM GSTtblOrderFood ofd
            INNER JOIN GSTtblFoodMenu fm ON ofd.FoodMenuID = fm.FoodMenuID

            UNION ALL
            SELECT bp.BookingCode,
                   bp.Fine
            FROM GSTtblBookingPayment bp
            WHERE bp.Fine > 0

            UNION ALL
            SELECT pe.BookingCode,
                   (e.Charges * pe.NoOfGuest)
            FROM GSTtblProvideEvent pe
            INNER JOIN GSTtblEvent e ON pe.EventID = e.EventID

    (
        SELECT BookingCode, SUM(PaidAmt) AS PaidAmount
        FROM GSTtblBookingPayment
        GROUP BY BookingCode

        ROW_NUMBER() OVER (ORDER BY ISNULL(bp.InvoiceCode,'') DESC) AS SrNo,
        ISNULL(bp.InvoiceCode,'-') AS InvoiceCode,

        ISNULL(CONVERT(VARCHAR(10), bp.PaymentDate,103), '-') AS PaymentDate,
        ISNULL(pm.PaymentMethodType,'-') AS PaymentMethodType,

        N'₹' + FORMAT((st.GrandTotal - ISNULL(ot.OfferAmount,0)), 'N2') AS TotalAmount,
        ISNULL(
            N'₹' + FORMAT(
                (st.GrandTotal - ISNULL(ot.OfferAmount,0))  
                - ISNULL((SELECT SUM(p2.PaidAmt)  
                          FROM GSTtblBookingPayment p2  
                          WHERE p2.BookingCode = b.BookingCode   
                            AND p2.InvoiceCode <= ISNULL(bp.InvoiceCode,'')),0), 'N2'
            ), '-') AS RemainingAmount,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0)) = ISNULL(tp.PaidAmount,0)
            THEN 'Paid'
            ELSE 'Pending'
        END AS Status,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0)) = ISNULL(tp.PaidAmount,0)
            THEN 22
            ELSE 4
        END AS StatusID

    INNER JOIN GSTtblGuest g ON b.GuestCode = g.GuestCode

    LEFT JOIN ServiceTotal st ON b.BookingCode = st.BookingCode
    LEFT JOIN OfferTotal ot ON b.BookingCode = ot.BookingCode
    LEFT JOIN TotalPaid tp ON b.BookingCode = tp.BookingCode
    WHERE 
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0)) = ISNULL(tp.PaidAmount,0)

            INNER JOIN GSTtblEvent e ON pe.EventID = e.EventID
        ) AS X
        GROUP BY X.BookingCode
    ),
    TotalPaid AS
    (
        SELECT BookingCode, SUM(PaidAmt) AS PaidAmount
        FROM GSTtblBookingPayment
        GROUP BY BookingCode
    )

    SELECT  
        ROW_NUMBER() OVER (ORDER BY ISNULL(bp.InvoiceCode,'') DESC) AS SrNo,
        ISNULL(bp.InvoiceCode,'-') AS InvoiceCode,
        b.BookingCode,
        g.GuestName,
        ISNULL(CONVERT(VARCHAR(10), bp.PaymentDate,103), '-') AS PaymentDate,
        ISNULL(pm.PaymentMethodType,'-') AS PaymentMethodType,
        ISNULL(N'₹' + FORMAT(bp.PaidAmt, 'N2'), '-') AS PaidAmount,
        N'₹' + FORMAT((st.GrandTotal - ISNULL(ot.OfferAmount,0)), 'N2') AS TotalAmount,
        ISNULL(
            N'₹' + FORMAT(
                (st.GrandTotal - ISNULL(ot.OfferAmount,0))  
                - ISNULL((SELECT SUM(p2.PaidAmt)  
                          FROM GSTtblBookingPayment p2  
                          WHERE p2.BookingCode = b.BookingCode   
                            AND p2.InvoiceCode <= ISNULL(bp.InvoiceCode,'')),0), 'N2'
            ), '-') AS RemainingAmount,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0))  
                 = ISNULL((SELECT SUM(p2.PaidAmt)  
                           FROM GSTtblBookingPayment p2  
                           WHERE p2.BookingCode = b.BookingCode),0)
            THEN 'Paid'
            ELSE 'Pending'
        END AS Status,
        CASE 
            WHEN (st.GrandTotal - ISNULL(ot.OfferAmount,0))  
                 = ISNULL((SELECT SUM(p2.PaidAmt)  
                           FROM GSTtblBookingPayment p2  
                           WHERE p2.BookingCode = b.BookingCode),0)
            THEN 22
            ELSE 4
        END AS StatusID

    FROM GSTtblBooking b
    INNER JOIN GSTtblGuest g ON b.GuestCode = g.GuestCode
    LEFT JOIN GSTtblBookingPayment bp ON b.BookingCode = bp.BookingCode  
    LEFT JOIN GSTtblPaymentMethod pm ON bp.PaymentMethodID = pm.PaymentMethodID
    LEFT JOIN ServiceTotal st ON b.BookingCode = st.BookingCode
    LEFT JOIN OfferTotal ot ON b.BookingCode = ot.BookingCode
    LEFT JOIN TotalPaid tp ON b.BookingCode = tp.BookingCode
    WHERE CONVERT(date,PaymentDate)BETWEEN @FromDateSP AND @ToDateSP  

    ORDER BY ISNULL(bp.InvoiceCode,'') DESC;

END

            FROM GSTtblProvideInventory pi
            INNER JOIN GSTtblInventory i ON pi.InventoryID = i.InventoryID
            WHERE pi.BookingCode = b.BookingCode
        ), '--'

            INNER JOIN GSTtblAddParkingSlot ps 
                ON pp.ParkingSlotID = ps.ParkingSlotID

 WHERE b.CheckIn BETWEEN @FromDateAP AND @ToDateAP

GROUP BY 
      b.BookingCode, 
      g.GuestName,
      r.RoomName,
      b.CheckIn, 
      b.CheckOut
ORDER BY b.BookingCode


end

    FROM GSTtblInventory i

    WHERE i.SubTypeID = 14;

------------- Auto low stock status Update ----------------

IF(@Flag = 'AutoLowStockStatusVK')
BEGIN

    DECLARE @Quantity INT
    DECLARE @LowStock INT

    -- GET CURRENT INVENTORY QUANTITY
    SELECT 
        @Quantity = Quantity
    FROM GSTtblInventory
    WHERE InventoryID = @InventoryIDVK

    -- GET LOW STOCK CRITERIA
    SELECT 
        @LowStock = s.LowStock
    FROM GSTtblInventory i
    INNER JOIN GSTtblInventoryStock s
        ON i.InventoryTypeID = s.InventoryID
    WHERE i.InventoryID = @InventoryIDVK

    -- CHECK LOW STOCK
    IF(@Quantity <= @LowStock)
    BEGIN
        -- LOW STOCK
        UPDATE GSTtblInventory
        SET StatusID = 14
        WHERE InventoryID = @InventoryIDVK
    END
    ELSE
    BEGIN
        -- IN STOCK
        UPDATE GSTtblInventory
        SET StatusID = 13
        WHERE InventoryID = @InventoryIDVK
    END
END

BEGIN
SELECT 
    S.ParkingSlotID,
    S.SlotName ,
    S.Capacity,

    -- UsedCount (ACTIVE BOOKINGS ONLY)
    (
        SELECT COUNT(*) 
        FROM GSTtblProvideParking P
        INNER JOIN GSTtblBooking B ON P.ProvideParkingID = B.ProvideParkingID
        WHERE 
            P.ParkingSlotID = S.ParkingSlotID
            AND GETDATE() >= CAST(B.Checkin AS DATETIME) + CAST(B.CheckinTime AS DATETIME)
            AND GETDATE() <= CAST(B.Checkout AS DATETIME) + CAST(B.CheckoutTime AS DATETIME)
    ) AS UsedCount,

    -- Slot Status
    CASE 
        WHEN (
            SELECT COUNT(*) 
            FROM GSTtblProvideParking P
            INNER JOIN GSTtblBooking B ON P.ProvideParkingID = B.ProvideParkingID
            WHERE 
                P.ParkingSlotID = S.ParkingSlotID
                AND GETDATE() >= CAST(B.Checkin AS DATETIME) + CAST(B.CheckinTime AS DATETIME)
                AND GETDATE() <= CAST(B.Checkout AS DATETIME) + CAST(B.CheckoutTime AS DATETIME)
        ) >= S.Capacity 
        THEN 'Occupied'
        ELSE 'Available'
    END AS SlotStatus

FROM GSTtblAddParkingSlot S;
END

BEGIN
    BEGIN TRY
        BEGIN TRAN;

        -- UPDATE BOOKING (CHECKOUT)
        UPDATE GSTtblBooking
        SET
            Checkout = @CheckOutFL,
            CheckoutTime = @CheckOutTimeFL
        WHERE BookingCode = @BookingCodeFL;

        -- Ensure row affected
        IF @@ROWCOUNT = 0
            THROW 50001, 'Booking not found', 1;

        -- ROOM DIRTY (trim spaces and case-insensitive)
        UPDATE GSTtblRoom
        SET statusID = 19
        WHERE LOWER(LTRIM(RTRIM(RoomName))) = LOWER(LTRIM(RTRIM(@RoomNameFL)));

        IF @@ROWCOUNT = 0
            THROW 50002, 'Room not found', 1;

        COMMIT TRAN;

        SELECT 'SUCCESS' AS Result;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRAN;

        SELECT 
            'FAIL' AS Result,
            ERROR_NUMBER() AS ErrorNumber,
            ERROR_MESSAGE() AS ErrorMessage;
    END CATCH
END

-------------------- AUTO ROOM STATUS UPDATE --------------------------
IF @FlagFL = 'AutoRoomStatusFL'
BEGIN

    ------------------------------------------------
    -- STEP 1 : DIRTY (19) -> CLEANING (18)
    -- AFTER 2 MINUTES FROM CHECKOUT
    ------------------------------------------------
    UPDATE r
    SET r.StatusID = 18
    FROM GSTtblRoom r
    INNER JOIN GSTtblBooking b
        ON r.RoomID = b.RoomID
    WHERE r.StatusID = 19
      AND b.BookingID =
      (
          SELECT MAX(BookingID)
          FROM GSTtblBooking
          WHERE RoomID = r.RoomID
      )
      AND b.CheckOut IS NOT NULL
      AND b.CheckoutTime IS NOT NULL
      AND DATEDIFF
      (
          SECOND,
          DATEADD
          (
              SECOND,
              DATEDIFF(SECOND, '00:00:00', b.CheckoutTime),
              CAST(b.CheckOut AS DATETIME)
          ),
          GETDATE()
      ) >= 120;



    ------------------------------------------------
    -- STEP 2 : CLEANING (18) -> AVAILABLE (16)
    -- AFTER 5 MINUTES FROM CHECKOUT
    ------------------------------------------------
    UPDATE r
    SET r.StatusID = 16
    FROM GSTtblRoom r
    INNER JOIN GSTtblBooking b
        ON r.RoomID = b.RoomID
    WHERE r.StatusID = 18
      AND b.BookingID =
      (
          SELECT MAX(BookingID)
          FROM GSTtblBooking
          WHERE RoomID = r.RoomID
      )
      AND b.CheckOut IS NOT NULL
      AND b.CheckoutTime IS NOT NULL
      AND DATEDIFF
      (
          SECOND,
          DATEADD
          (
              SECOND,
              DATEDIFF(SECOND, '00:00:00', b.CheckoutTime),
              CAST(b.CheckOut AS DATETIME)
          ),
          GETDATE()
      ) >= 300;

END

IF @FlagFL = 'AddBookingFL'
BEGIN
    BEGIN TRY
        BEGIN TRAN;

        DECLARE @NextBookingCodeFL VARCHAR(20);
        DECLARE @BookingID INT;
        DECLARE @NewProvideParkingID INT;

        -- BOOKING CODE
        SELECT @NextBookingCodeFL =
            'BK' + CAST(
                ISNULL(MAX(CAST(REPLACE(BookingCode,'BK','') AS BIGINT)),0) + 1
            AS VARCHAR(20))
        FROM GSTtblBooking
        WHERE BookingCode LIKE 'BK%';

        -- INSERT BOOKING
        INSERT INTO GSTtblBooking
        (
            BookingCode, GuestCode, RoomID, Checkin,
            CheckinTime, Checkout, CheckoutTime,
            No_of_Guest, BookingDate, AgentID, OfferID
        )
        VALUES
        (
            @NextBookingCodeFL, @GuestCodeFL, @RoomIDFL,
            @CheckInFL, @CheckInTimeFL, @CheckOutFL, @CheckOutTimeFL,
            @NoOfGuestFL, @CheckInBFL, @AgentCodeFL, @OfferIDFL
        );

        SET @BookingID = SCOPE_IDENTITY();

        -- PARKING
        IF (@VehicleNumFL IS NOT NULL AND @VehicleNumFL <> '')
        BEGIN
            INSERT INTO GSTtblProvideParking
            (BookingCode, ParkingSlotID, VehicleNumber, VehicleTypeID)
            VALUES
            (@NextBookingCodeFL, @ParkingSlotIDFL, @VehicleNumFL, @VehicleTypeIDFL);

            SET @NewProvideParkingID = SCOPE_IDENTITY();

            UPDATE GSTtblBooking
            SET ProvideParkingID = @NewProvideParkingID
            WHERE BookingID = @BookingID;
        END

        -- ROOM OCCUPIED
        UPDATE GSTtblRoom
        SET statusID = 12
        WHERE RoomID = @RoomIDFL;

        COMMIT TRAN;

        SELECT @NextBookingCodeFL AS NewBookingCodeFL;
    END TRY
    BEGIN CATCH
        ROLLBACK;
        THROW;
    END CATCH
END
